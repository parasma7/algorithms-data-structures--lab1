#include <iostream>
using namespace std;

/**
 * Функция проверяет наличие одинаковых цифр в числе.
 *
 * @param number Число для проверки (целое).
 * @return true — если хотя бы одна цифра повторяется; false — иначе.
 */
bool hasDuplicateDigits(int number) {
    // Создаём массив из 10 элементов, чтобы отмечать появление каждой цифры от 0 до 9.
    bool digits[10] = {false};
    
    while (number != 0) {
        int digit = abs(number % 10); // Берём последнюю цифру (модуль нужен для отрицательных чисел)
        
        // Если такая цифра уже встречалась ранее, возвращаем истину.
        if (digits[digit]) return true;
        
        // Отмечаем текущую цифру как найденную.
        digits[digit] = true;
        
        // Убираем последнюю цифру из числа.
        number /= 10;
    }
    return false;
}

int main() {
    double a, b, c;
    char symbol;

    /* Блок ввода данных */
    cout << "Введите коэффициенты многочлена ax²+bx+c через пробел: ";
    cin >> a >> b >> c;

    cout << "Введите символ S: ";
    cin >> symbol;

    /* Блок обработки */
    // Для поиска одинаковых цифр нам нужно целое число без дробной части.
    // Используем коэффициент 'a', так как он стоит первым по условию.
    int integerPart = static_cast<int>(a);

    // Вызываем функцию проверки наличия дубликатов.
    bool result = hasDuplicateDigits(integerPart);

    /* Блок вывода результата */
    if (result) 
        cout << "Среди цифр числа есть одинаковые." << endl;
    else 
        cout << "Все цифры числа разные." << endl;

    return 0;
}
