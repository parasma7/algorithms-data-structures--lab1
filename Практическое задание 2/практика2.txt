Задание 4

#include <iostream>
#include <vector>   // Для удобных динамических массивов (векторов)
#include <algorithm> // Для функции сортировки

int main() {
    // Наши исходные списки чисел
    int first[] = {1, 3, 5, 7};
    int second[] = {2, 4, 6, 8, 10};

    // Создаем пустой список для результата.
    // Вектор удобен тем, что сам увеличивает свой размер,
    // когда мы добавляем в него новые числа.
    std::vector<int> result;

    // Проходим по первому массиву и перекладываем всё в результат.
    // Цикл вида "for (элемент : массив)" берет каждое число по очереди.
    for (int num : first) {
        result.push_back(num); // Добавляем текущее число в конец вектора
    }

    // То же самое делаем со вторым массивом
    for (int num : second) {
        result.push_back(num);
    }

    // Сортируем от меньшего к большему.
    // Нам не пришлось писать свою логику сравнения — используем готовую функцию.
    std::sort(result.begin(), result.end());

    std::cout << "Слитый массив: ";
    
    // а затем выводим все числа из готового вектора через пробел
    for (int num : result) {
        std::cout << num << " ";
    }
    
    // Переносим курсор на новую строку после вывода всех данных
    std::cout << std::endl;

    return 0; // Говорим системе, что программа завершилась успешно
}

#include <iostream>
#include <vector>

using namespace std;

int main() {
    // У нас есть ряд чисел от 0 до n, но одно число потерялось.
    // Например: должно быть [0, 1, 2, 3, 4, 5], а в массиве только пять элементов.
    vector<int> numbers = {0, 1, 2, 4, 5}; 
    
    // Смотрим, сколько чисел у нас на руках. 
    // Если их 5 (значит n = 5), то полный ряд должен был закончиться числом 5,
    // и всего чисел было бы 6 (от 0 до 5 включительно).
    int n = numbers.size(); 

    // Сначала посчитаем, какая сумма должна была получиться В ИДЕАЛЕ.
    // Для этого берем формулу суммы всех чисел от 1 до n: n * (n + 1) / 2.
    // Так как мы считаем от нуля, результат не изменится, просто прибавится "0".
    // Используем long long — это очень большие числа, чтобы программа не сломалась, 
    // если вдруг кто-то передаст сюда миллион элементов.
    long long ideal_sum = static_cast<long long>(n) * (n + 1) / 2;

    long long real_sum = 0; // Здесь будем копить сумму тех чисел, что реально есть

    // Пробегаемся по нашему массиву обычным циклом и складываем всё, что нашли
    for (int i = 0; i < n; ++i) {
        real_sum += numbers[i];
    }

    // Логика простая: если из того, что ДОЛЖНО БЫЛО быть, вычесть то, что ЕСТЬ,
    // останется ровно то самое потерянное число.
    int missing_number = ideal_sum - real_sum;

    cout << "Потерянное число: " << missing_number << endl;

    return 0;
}
