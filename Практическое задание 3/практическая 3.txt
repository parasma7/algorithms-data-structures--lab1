#include <iostream>
#include <cmath>
using namespace std;

double hypotenuse(double a, double b) {
    return sqrt(a * a + b * b);
}

int main() {
    double a, b;
    cout << "Введите катеты a и b: ";
    cin >> a >> b;
    
    cout << "Гипотенуза = " << hypotenuse(a, b) << endl;
    return 0;
}

''третий способ: ООП''

#include <iostream>
#include <cmath>
using namespace std;

class Triangle {
private:
    double cathetus_a = 0; 
    double cathetus_b = 0;
public:
    void input_sides() { 
        cout << "Введите два положительных катета a и b: ";
        cin >> cathetus_a >> cathetus_b;
        
        if (cathetus_a <= 0 || cathetus_b <= 0)
            throw runtime_error("Катеты должны быть положительными!");
    }
    
    double get_hypotenuse() const {
        return sqrt(cathetus_a * cathetus_a + cathetus_b * cathetus_b);  
    }
};

int main() {
    try {
        Triangle t;
        t.input_sides();   
        cout << "Гипотенуза = " << t.get_hypotenuse() << endl;
    } catch(const exception& e) {
        cerr << "Ошибка! " << e.what() << endl;
    }
    return 0;
}

''третий способ: ООП''

#include <iostream>
using namespace std;

class Number {
private:
    int value;   
public:
    void input() { 
        cout << "Введите число: "; 
        cin >> value; 
    }

    int getTensDigit() const {
        return (value / 10) % 10;
    }
};

int main() {
    Number num;
    num.input();                     
    cout << "Число десятков: " << num.getTensDigit() << endl;
    return 0;
}
